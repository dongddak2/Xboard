def check(name, actual, expected):
    if actual != expected:

def method_body(src, name):
    start = src.index("function %s(" % name)
    rest = src[start:]
    nxt = rest.find("\n    public ", 10)
    nxt2 = rest.find("\n    private ", 10)
    cut = min([x for x in (nxt, nxt2) if x != -1] or [len(rest)])
    return rest[:cut]
